\usepackage[skins,most]{tcolorbox}
\usetikzlibrary{calc,shadows.blur,decorations.pathreplacing}

% --- Couleurs spécifiques ---
\definecolor{eceyellow}{rgb}{0.8,0.5,0}
\definecolor{ECE}{RGB}{0, 122, 123}

% =====================================================================
%  RÉGLAGES DU RENDU DES QUESTIONS
% =====================================================================
%  Deux rendus pour la même sémantique :
%
%    \TQ{...}   \EQ{...}    rendu FILAIRE  (défaut) — questions enchaînées
%    \TQ*{...}  \EQ*{...}   rendu ENCADRÉ            — question isolée à
%                                                      mettre en valeur
%
%  Numérotation et couleurs identiques dans les deux cas : on peut
%  passer de l'un à l'autre en ajoutant/retirant une étoile, sans rien
%  casser.
% =====================================================================

% Police de l'énoncé en rendu filaire :
%   \bfseries   → énoncé en gras (style « sujet de TP » classique)
%   \normalfont → énoncé en romain, seul le numéro reste en gras
%                 (plus lisible quand les énoncés font 3-4 lignes)
\newcommand{\qfont}{\bfseries}

% Couleurs des deux familles de questions
\colorlet{TQcolor}{ECE}      % théorie      → T1, T2, ...
\colorlet{EQcolor}{gamboge}  % pratique     → E1, E2, ...

% --- Compteurs (réinitialisés à chaque chapitre) ---
\newcounter{tqcounter}[chapter]
\newcounter{eqcounter}[chapter]
\renewcommand{\thetqcounter}{T\arabic{tqcounter}}
\renewcommand{\theeqcounter}{E\arabic{eqcounter}}

% --- Espacement autour d'une question filaire ---
\newlength{\qskipbefore}\setlength{\qskipbefore}{1.6ex plus .4ex minus .2ex}
\newlength{\qskipafter} \setlength{\qskipafter} {1.6ex plus .3ex}

% Saut de paragraphe explicite (utile après un énoncé, une figure, etc.)
\newcommand{\n}{\par\addvspace{\qskipbefore}\noindent}

% Cœur du rendu filaire : #1 = label formaté, #2 = énoncé
% (\long : l'énoncé peut contenir plusieurs paragraphes, un tableau, etc.)
% Un filet vertical à la couleur de la famille (T ou E) borde la question à gauche :
% il sépare visuellement l'énoncé du texte de contexte qui le suit.
\newcommand{\qpar}[3]{%  #1 = couleur, #2 = label, #3 = énoncé
    \par\addvspace{\qskipbefore}%
    \begin{tcolorbox}[enhanced, breakable, blanker, left=7pt, top=1pt, bottom=1pt,
                      borderline west={1.5pt}{0pt}{#1}, before skip=0pt, after skip=0pt]
        \noindent\textcolor{#1}{#2}\hspace{0.5em}{\qfont #3}%
    \end{tcolorbox}%
    \par\addvspace{\qskipafter}%
}

% Style commun pour les boîtes de questions
\tcbset{
    qboxstyle/.style={
        enhanced,
        colback=yorhabg,
        coltext=yorhafg,
        coltitle=mwhite,
        attach boxed title to top left={yshift*=-\tcboxedtitleheight},
        boxed title size=title,
        boxed title style={
            rounded corners=northeast,
            rounded corners=northwest,
            colback=tcbcolframe,
            boxrule=0pt,
        },
        underlay boxed title={
            \path[fill=tcbcolframe]
                (title.south west) -- (title.south east)
                to[out=0,in=180] ([xshift=5mm]title.east) --
                (title.center-|frame.east) [rounded corners=5pt] |-
                (frame.north) -| cycle;
        },
        before skip=0.8em,
        after skip=0.3em,
    }
}

% Cœur du rendu encadré : #1 = couleur, #2 = titre, #3 = énoncé
\newcommand{\qboxed}[3]{%
    \begin{tcolorbox}[qboxstyle, colframe=#1, title=\textbf{#2}]
        #3
    \end{tcolorbox}%
}

% QUESTION BOX libre, sans numérotation (ex-\begin{question})
\newtcolorbox{questionbox}{
    qboxstyle,
    colframe=ECE,
    title=\texttt{Question.},
}

% =====================================================================
%  COMMANDES DE QUESTION
% =====================================================================
%  Syntaxe :  \TQ{énoncé}            filaire, numéroté T1, T2, ...
%             \TQ*{énoncé}           encadré, même numérotation
%             \TQ[clé]{énoncé}       pose un \label{clé} ; \ref{clé} → « T3 »
%
%  Idem pour \EQ (numérotation E1, E2, ...).
% =====================================================================

% THÉORIE
\NewDocumentCommand{\TQ}{s O{} +m}{%
    \refstepcounter{tqcounter}%
    \ifblank{#2}{}{\label{#2}}%
    \IfBooleanTF{#1}
        {\qboxed{TQcolor}{\thetqcounter.}{#3}}
        {\qpar{TQcolor}{\thetqcounter.}{#3}}%
}

% PRATIQUE
\NewDocumentCommand{\EQ}{s O{} +m}{%
    \refstepcounter{eqcounter}%
    \ifblank{#2}{}{\label{#2}}%
    \IfBooleanTF{#1}
        {\qboxed{EQcolor}{\theeqcounter.}{#3}}
        {\qpar{EQcolor}{\theeqcounter.}{#3}}%
}

% Question libre, non numérotée
%   \question{énoncé}   filaire
%   \question*{énoncé}  encadré (équivalent de \begin{questionbox})
\NewDocumentCommand{\question}{s +m}{%
    \IfBooleanTF{#1}
        {\qboxed{ECE}{Question.}{#2}}
        {\qpar{ECE}{Question.}{#2}}%
}
