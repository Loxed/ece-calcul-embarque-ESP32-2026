\tcbset{
    qboxstyle/.style={
        enhanced,
        colback=yorhabg,
        coltext=yorhafg,
        coltitle=mwhite,
        attach boxed title to top left={yshift*=-\tcboxedtitleheight},
        boxed title size=title,
        boxed title style={
            rounded corners=northeast,
            rounded corners=northwest,
            colback=tcbcolframe,
            boxrule=0pt,
        },
        underlay boxed title={
            \path[fill=tcbcolframe] 
                (title.south west) -- (title.south east) 
                to[out=0,in=180] ([xshift=5mm]title.east) -- 
                (title.center-|frame.east) [rounded corners=5pt] |- 
                (frame.north) -| cycle;
        },
        before skip=0.8em,
        after skip=0.3em,
    }
}

% THÉORIE (Commande \TQ{content})
\newcommand{\TQ}[1]{%
    \stepcounter{tqcounter}%
    \begin{tcolorbox}[qboxstyle, colframe=ECE, title=\textbf{T\arabic{tqcounter}.}]
        #1
    \end{tcolorbox}%
}

% PRATIQUE (Commande \EQ{content})
\newcommand{\EQ}[1]{%
    \stepcounter{eqcounter}%
    \begin{tcolorbox}[qboxstyle, colframe=gamboge, title=\textbf{E\arabic{eqcounter}.}]
        #1
    \end{tcolorbox}%
}
